# Copy to an ignored build directory as CMakeLists.txt. LLAMA_SOURCE and
# LLAMA_BUILD identify the pinned independent checkout/build.
cmake_minimum_required(VERSION 3.21)
project(gemma4_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_executable(llama-reference "${GUFO_SOURCE}/src/models/gemma4/tools/llama_reference.cpp")
add_executable(llama-assistant-reference "${GUFO_SOURCE}/src/models/gemma4/tools/llama_assistant_reference.cpp")
add_executable(llama-vision-reference "${GUFO_SOURCE}/src/models/gemma4/tools/llama_vision_reference.cpp")
add_executable(llama-image-reference "${GUFO_SOURCE}/src/models/gemma4/tools/llama_image_reference.cpp")
add_executable(llama-performance-reference "${GUFO_SOURCE}/src/models/gemma4/tools/llama_performance_reference.cpp")
foreach(target IN ITEMS llama-reference llama-assistant-reference llama-vision-reference llama-image-reference llama-performance-reference)
  target_include_directories(${target} PRIVATE "${LLAMA_SOURCE}/include" "${LLAMA_SOURCE}/ggml/include" "${LLAMA_SOURCE}/src")
  if(WIN32)
    target_link_libraries(${target} PRIVATE "${LLAMA_BUILD}/src/llama.lib" "${LLAMA_BUILD}/ggml/src/ggml-base.lib")
  else()
    target_link_libraries(${target} PRIVATE "${LLAMA_BUILD}/src/libllama.so" "${LLAMA_BUILD}/ggml/src/libggml-base.so")
  endif()
endforeach()
target_include_directories(llama-vision-reference PRIVATE "${LLAMA_SOURCE}/tools/mtmd")
if(WIN32)
  target_link_libraries(llama-vision-reference PRIVATE "${LLAMA_BUILD}/tools/mtmd/mtmd.lib")
else()
  target_link_libraries(llama-vision-reference PRIVATE "${LLAMA_BUILD}/tools/mtmd/libmtmd.so")
endif()
